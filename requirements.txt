# and BLAS versions. See PRD NF4 (determinism).

httpx>=0.27

# --- Phase 3: agent layer ---------------------------------------------------
langgraph>=1.0            # orchestration graph (confirmed on Python 3.14)
langchain-core>=1.0       # pulled in by langgraph
# LLM calls go through OpenRouter (OpenAI-compatible) over httpx — see
# app/agents/llm.py. No vendor SDK is required; httpx (Phase 1b) is reused.
# A paid vendor SDK can be swapped in later behind the same StructuredLLM seam.

# --- Phase 4: report generation ---------------------------------------------
# The Markdown and HTML renderers are pure Python, and the 8 charts are
# hand-generated inline SVG (no Jinja2, no Plotly) — inline SVG renders in both
# a browser and a PDF engine and is deterministic/testable.
# weasyprint is OPTIONAL: it renders the HTML to PDF only where its native
# libraries (Pango/Cairo/GObject) are present (Linux/CI). On a bare Windows box
# it cannot import, and the app falls back to browser "Print -> Save as PDF".
weasyprint>=62.0         # optional; HTML -> PDF where native libs exist

# --- Phase 5: application ---------------------------------------------------
fastapi>=0.111
uvicorn>=0.30            # plain (not [standard]) — uvloop has no Windows wheel
# Server-sent-event progress uses Starlette's StreamingResponse (no sse-starlette),
# history persistence uses the standard-library sqlite3 (no SQLAlchemy), and the
# browser UI is a self-contained page served by FastAPI (no Streamlit) — keeping
# the dependency surface and the test story small.
